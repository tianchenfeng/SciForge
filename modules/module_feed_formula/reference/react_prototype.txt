import React, { useState, useEffect } from 'react';
import { PieChart, Pie, Cell, Tooltip, ResponsiveContainer } from 'recharts';
import { Calculator, AlertCircle, CheckCircle, Plus, Trash2, Save, RotateCcw } from 'lucide-react';

// 动态加载外部脚本的 Hook
const useScript = (url) => {
  const [loaded, setLoaded] = useState(false);
  useEffect(() => {
    const script = document.createElement('script');
    script.src = url;
    script.async = true;
    script.onload = () => setLoaded(true);
    document.body.appendChild(script);
    return () => { document.body.removeChild(script); };
  }, [url]);
  return loaded;
};

// 空白模板，方便用户添加
const EMPTY_ROW = { 
    id: Date.now(), name: '新原料', type: 'concentrate', 
    cost: 0, dm: 88, cp: 0, starch: 0, ndf: 0, adf: 0, ca: 0, p: 0 
};

// 默认显示的示例（用户可以清空）
const DEFAULT_ROWS = [
    { id: 1, name: '原料A (粗)', type: 'roughage', cost: 2.5, dm: 90, cp: 18, starch: 1.2, ndf: 42, adf: 30, ca: 1.4, p: 0.25 },
    { id: 2, name: '原料B (精)', type: 'concentrate', cost: 3.2, dm: 89, cp: 46.5, starch: 5.0, ndf: 12, adf: 5, ca: 0.35, p: 0.65 },
];

const COLORS = ['#0088FE', '#00C49F', '#FFBB28', '#FF8042', '#8884d8', '#82ca9d', '#ffc658'];

export default function App() {
    const solverLoaded = useScript("https://unpkg.com/javascript-lp-solver/prod/solver.js");

    // --- 状态管理 ---
    
    // 1. 原料表数据 (完全可编辑)
    const [ingredients, setIngredients] = useState(DEFAULT_ROWS);

    // 2. 约束条件 (营养 + 精粗比)
    const [constraints, setConstraints] = useState({
        cp: { label: '粗蛋白 (CP)', enable: true, target: 17.5, error: 1.0 },
        starch: { label: '淀粉 (Starch)', enable: true, target: 24.0, error: 2.0 },
        ndf: { label: 'NDF', enable: false, target: 30.0, error: 2.0 },
        adf: { label: 'ADF', enable: false, target: 19.0, error: 2.0 },
        ca: { label: '钙 (Ca)', enable: false, target: 0.6, error: 0.1 },
        p: { label: '磷 (P)', enable: false, target: 0.4, error: 0.05 },
        // 精粗比作为一个特殊的约束存在
        ratio: { label: '粗饲料占比 (Roughage %)', enable: true, target: 50.0, error: 5.0 }
    });

    const [result, setResult] = useState(null);
    const [status, setStatus] = useState('idle');

    // --- 操作函数 ---

    const addRow = () => {
        setIngredients([...ingredients, { ...EMPTY_ROW, id: Date.now() }]);
    };

    const deleteRow = (id) => {
        setIngredients(ingredients.filter(i => i.id !== id));
    };

    const updateRow = (id, field, value) => {
        setIngredients(ingredients.map(row => 
            row.id === id ? { ...row, [field]: value } : row
        ));
    };

    const updateConstraint = (key, field, value) => {
        setConstraints(prev => ({
            ...prev,
            [key]: { ...prev[key], [field]: value }
        }));
    };

    // --- 求解核心逻辑 ---
    const solveOptimization = () => {
        if (!solverLoaded) return;
        setStatus('solving');

        // 1. 构建 LP 模型
        const model = {
            optimize: "cost",
            opType: "min",
            constraints: {
                totalWeight: { equal: 100 }, // 强制总量 100%
            },
            variables: {}
        };

        // 2. 添加所有启用的约束 (包括营养 和 精粗比)
        Object.entries(constraints).forEach(([key, setting]) => {
            if (setting.enable) {
                const min = parseFloat(setting.target) - parseFloat(setting.error);
                const max = parseFloat(setting.target) + parseFloat(setting.error);
                
                // 这里的 key 需要和下面 variables 里的 key 对应
                // 注意：精粗比的 key 是 'ratio'，但我们在 variables 里统计的是 'roughageWeight'
                const constraintKey = key === 'ratio' ? 'roughageWeight' : key;
                
                model.constraints[constraintKey] = { min: Math.max(0, min), max: max };
            }
        });

        // 3. 定义变量 (即每种原料)
        ingredients.forEach(ing => {
            // 只有当原料有名字时才参与计算
            if (!ing.name.trim()) return;

            // 处理数据类型，确保是数字
            const safeParse = (val) => isNaN(parseFloat(val)) ? 0 : parseFloat(val);

            model.variables[ing.name] = {
                totalWeight: 1, // 自身重量贡献
                cost: safeParse(ing.cost),
                // 营养贡献
                cp: safeParse(ing.cp),
                starch: safeParse(ing.starch),
                ndf: safeParse(ing.ndf),
                adf: safeParse(ing.adf),
                ca: safeParse(ing.ca),
                p: safeParse(ing.p),
                // 精粗比贡献：如果是粗料，则贡献1单位的粗料重量
                roughageWeight: ing.type === 'roughage' ? 1 : 0,
                
                // 自身标记，用于输出结果
                [ing.name]: 1
            };
        });

        // 4. 执行求解
        try {
            const solution = window.solver.Solve(model);
            
            if (!solution.feasible) {
                setStatus('error');
                setResult({ message: "无解。请尝试放宽误差范围，或检查原料数据是否合理。" });
            } else {
                setStatus('solved');
                
                // 整理结果
                const finalMix = [];
                let stats = { cost: 0, roughage: 0, cp: 0, starch: 0, ndf: 0, adf: 0, ca: 0, p: 0 };

                ingredients.forEach(ing => {
                    if (solution[ing.name]) {
                        const amount = solution[ing.name];
                        if (amount > 0.001) {
                            finalMix.push({ ...ing, amount });
                            
                            // 累加统计
                            stats.cost += amount * ing.cost;
                            if (ing.type === 'roughage') stats.roughage += amount;
                            stats.cp += amount * ing.cp / 100;
                            stats.starch += amount * ing.starch / 100;
                            stats.ndf += amount * ing.ndf / 100;
                            stats.adf += amount * ing.adf / 100;
                            stats.ca += amount * ing.ca / 100;
                            stats.p += amount * ing.p / 100;
                        }
                    }
                });

                setResult({ mix: finalMix, stats });
            }
        } catch (e) {
            setStatus('error');
            setResult({ message: "运算错误: " + e.message });
        }
    };

    return (
        <div className="min-h-screen bg-gray-100 p-4 font-sans text-gray-800">
            <div className="max-w-7xl mx-auto space-y-6">
                
                {/* 顶部标题栏 */}
                <div className="flex justify-between items-center bg-white p-4 rounded-lg shadow-sm border border-gray-200">
                    <div>
                        <h1 className="text-2xl font-bold text-gray-800">饲料配方计算器 (自由版)</h1>
                        <p className="text-xs text-gray-500">Powered by Tianchenfeng</p>
                    </div>
                    <div className="flex gap-2">
                        <button onClick={() => setIngredients(DEFAULT_ROWS)} className="text-xs text-gray-500 hover:text-indigo-600 flex items-center gap-1">
                            <RotateCcw size={14}/> 重置示例
                        </button>
                    </div>
                </div>

                <div className="grid grid-cols-1 xl:grid-cols-12 gap-6">
                    
                    {/* 左侧：数据输入区 (占大头) */}
                    <div className="xl:col-span-8 space-y-6">
                        
                        {/* 1. 原料录入表 */}
                        <div className="bg-white rounded-lg shadow-sm border border-gray-200 overflow-hidden">
                            <div className="p-4 border-b border-gray-100 flex justify-between items-center bg-gray-50">
                                <h3 className="font-bold text-gray-700">1. 原料成分录入 (干物质基础 %)</h3>
                                <button onClick={addRow} className="bg-indigo-600 hover:bg-indigo-700 text-white text-sm px-3 py-1.5 rounded flex items-center gap-1 transition-colors">
                                    <Plus size={16} /> 添加原料
                                </button>
                            </div>
                            
                            <div className="overflow-x-auto">
                                <table className="w-full text-sm text-left">
                                    <thead className="bg-gray-100 text-gray-600 font-medium border-b border-gray-200">
                                        <tr>
                                            <th className="p-3 min-w-[120px]">原料名称</th>
                                            <th className="p-3 min-w-[100px]">类型</th>
                                            <th className="p-3 w-20 text-center">价格</th>
                                            <th className="p-3 w-16 text-center text-gray-400" title="仅供参考，不参与计算">DM%</th>
                                            <th className="p-3 w-20 text-center text-indigo-700 bg-indigo-50">CP</th>
                                            <th className="p-3 w-20 text-center text-indigo-700 bg-indigo-50">淀粉</th>
                                            <th className="p-3 w-20 text-center">NDF</th>
                                            <th className="p-3 w-20 text-center">ADF</th>
                                            <th className="p-3 w-20 text-center">Ca</th>
                                            <th className="p-3 w-20 text-center">P</th>
                                            <th className="p-3 w-12 text-center">操作</th>
                                        </tr>
                                    </thead>
                                    <tbody className="divide-y divide-gray-100">
                                        {ingredients.map((row) => (
                                            <tr key={row.id} className="hover:bg-gray-50 group transition-colors">
                                                <td className="p-2">
                                                    <input 
                                                        type="text" value={row.name} 
                                                        onChange={(e) => updateRow(row.id, 'name', e.target.value)}
                                                        className="w-full border-gray-300 rounded px-2 py-1 focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500 border"
                                                        placeholder="名称"
                                                    />
                                                </td>
                                                <td className="p-2">
                                                    <select 
                                                        value={row.type}
                                                        onChange={(e) => updateRow(row.id, 'type', e.target.value)}
                                                        className={`w-full border rounded px-2 py-1 text-xs font-medium ${row.type === 'roughage' ? 'text-green-700 bg-green-50 border-green-200' : 'text-amber-700 bg-amber-50 border-amber-200'}`}
                                                    >
                                                        <option value="concentrate">精饲料</option>
                                                        <option value="roughage">粗饲料</option>
                                                    </select>
                                                </td>
                                                {['cost', 'dm', 'cp', 'starch', 'ndf', 'adf', 'ca', 'p'].map(field => (
                                                    <td key={field} className={`p-2 ${['cp','starch'].includes(field) ? 'bg-indigo-50/30' : ''}`}>
                                                        <input 
                                                            type="number" step="0.01" 
                                                            value={row[field]}
                                                            onChange={(e) => updateRow(row.id, field, e.target.value)}
                                                            className="w-full text-center border-gray-200 bg-transparent rounded px-1 py-1 focus:bg-white focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500 border"
                                                        />
                                                    </td>
                                                ))}
                                                <td className="p-2 text-center">
                                                    <button onClick={() => deleteRow(row.id)} className="text-gray-400 hover:text-red-500 transition-colors">
                                                        <Trash2 size={16} />
                                                    </button>
                                                </td>
                                            </tr>
                                        ))}
                                    </tbody>
                                </table>
                            </div>
                            {ingredients.length === 0 && (
                                <div className="p-8 text-center text-gray-400 bg-gray-50 border-t border-dashed">
                                    暂无原料，请点击上方“添加原料”
                                </div>
                            )}
                        </div>

                        {/* 2. 约束条件设定 */}
                        <div className="bg-white rounded-lg shadow-sm border border-gray-200 p-5">
                            <h3 className="font-bold text-gray-700 mb-4">2. 设定配方要求 (Target ± Error)</h3>
                            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                                {Object.entries(constraints).map(([key, setting]) => (
                                    <div key={key} className={`border rounded-lg p-3 transition-all ${setting.enable ? 'border-indigo-300 bg-indigo-50/50 shadow-sm' : 'border-gray-200 bg-gray-50 opacity-60'}`}>
                                        <div className="flex items-center justify-between mb-2">
                                            <label className="font-semibold text-sm text-gray-700 flex items-center gap-2 cursor-pointer">
                                                <input 
                                                    type="checkbox" 
                                                    checked={setting.enable}
                                                    onChange={(e) => updateConstraint(key, 'enable', e.target.checked)}
                                                    className="w-4 h-4 text-indigo-600 rounded focus:ring-indigo-500"
                                                />
                                                {setting.label}
                                            </label>
                                        </div>
                                        
                                        <div className="flex items-center gap-2">
                                            <div className="relative flex-1">
                                                <span className="absolute left-2 top-1.5 text-xs text-gray-400 font-mono">目</span>
                                                <input 
                                                    type="number" step="0.1"
                                                    value={setting.target}
                                                    onChange={(e) => updateConstraint(key, 'target', e.target.value)}
                                                    disabled={!setting.enable}
                                                    className="w-full pl-6 pr-1 py-1 text-sm border border-gray-300 rounded bg-white focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500 disabled:bg-gray-100"
                                                />
                                            </div>
                                            <span className="text-gray-400">±</span>
                                            <div className="relative w-20">
                                                <span className="absolute left-2 top-1.5 text-xs text-gray-400 font-mono">误</span>
                                                <input 
                                                    type="number" step="0.1"
                                                    value={setting.error}
                                                    onChange={(e) => updateConstraint(key, 'error', e.target.value)}
                                                    disabled={!setting.enable}
                                                    className="w-full pl-6 pr-1 py-1 text-sm border border-gray-300 rounded bg-white focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500 disabled:bg-gray-100"
                                                />
                                            </div>
                                        </div>
                                    </div>
                                ))}
                            </div>
                        </div>

                        {/* 计算按钮 */}
                        <button 
                            onClick={solveOptimization}
                            className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-4 rounded-lg shadow-lg active:scale-[0.99] transition-all flex items-center justify-center gap-2 text-lg"
                        >
                            <Calculator size={24} /> 
                            开始计算 (Solve)
                        </button>
                    </div>

                    {/* 右侧：结果面板 */}
                    <div className="xl:col-span-4">
                        <div className="bg-white rounded-lg shadow-sm border border-gray-200 h-full flex flex-col p-5 sticky top-6">
                            <h3 className="font-bold text-gray-700 mb-4 border-b pb-2">计算结果</h3>
                            
                            {status === 'idle' && <div className="text-gray-400 text-center py-10">等待计算...</div>}
                            
                            {status === 'error' && (
                                <div className="bg-red-50 text-red-700 p-4 rounded border border-red-200 flex gap-2">
                                    <AlertCircle className="shrink-0 mt-0.5" size={18}/>
                                    <div className="text-sm">{result?.message}</div>
                                </div>
                            )}

                            {status === 'solved' && result && (
                                <div className="space-y-6 animate-in fade-in slide-in-from-bottom-2">
                                    
                                    {/* 饼图 */}
                                    <div className="h-48">
                                        <ResponsiveContainer width="100%" height="100%">
                                            <PieChart>
                                                <Pie
                                                    data={result.mix}
                                                    cx="50%"
                                                    cy="50%"
                                                    innerRadius={40}
                                                    outerRadius={65}
                                                    paddingAngle={2}
                                                    dataKey="amount"
                                                    nameKey="name"
                                                >
                                                    {result.mix.map((entry, index) => (
                                                        <Cell key={`cell-${index}`} fill={COLORS[index % COLORS.length]} stroke="none" />
                                                    ))}
                                                </Pie>
                                                <Tooltip />
                                            </PieChart>
                                        </ResponsiveContainer>
                                    </div>

                                    {/* 配方清单 */}
                                    <div>
                                        <div className="flex justify-between text-xs text-gray-500 mb-1 uppercase font-bold">
                                            <span>原料</span>
                                            <span>DM比例</span>
                                        </div>
                                        <div className="border rounded divide-y divide-gray-100 max-h-[300px] overflow-y-auto">
                                            {result.mix.map((item, idx) => (
                                                <div key={idx} className="flex justify-between p-2 hover:bg-gray-50 text-sm">
                                                    <div className="flex items-center gap-2">
                                                        <span className="w-2 h-2 rounded-full" style={{backgroundColor: COLORS[idx % COLORS.length]}}></span>
                                                        <span className="text-gray-800 font-medium">{item.name}</span>
                                                    </div>
                                                    <span className="font-bold text-indigo-600">{item.amount.toFixed(2)}%</span>
                                                </div>
                                            ))}
                                        </div>
                                        <div className="flex justify-between p-2 bg-gray-100 rounded mt-2 text-sm font-bold text-gray-700">
                                            <span>合计</span>
                                            <span>100%</span>
                                        </div>
                                    </div>

                                    {/* 营养验算 */}
                                    <div className="bg-gray-50 p-4 rounded-lg border border-gray-100">
                                        <h4 className="text-xs font-bold text-gray-500 uppercase mb-3">营养成分验算</h4>
                                        <div className="space-y-2 text-sm">
                                            <ResultRow label="成本 (Cost)" value={result.stats.cost.toFixed(2)} unit="元" />
                                            <ResultRow label="粗料占比" value={result.stats.roughage.toFixed(2)} unit="%" 
                                                isWarn={Math.abs(result.stats.roughage - constraints.ratio.target) > constraints.ratio.error} 
                                            />
                                            <ResultRow label="粗蛋白 (CP)" value={result.stats.cp.toFixed(2)} unit="%" 
                                                isWarn={constraints.cp.enable && Math.abs(result.stats.cp - constraints.cp.target) > constraints.cp.error}
                                            />
                                            <ResultRow label="淀粉 (Starch)" value={result.stats.starch.toFixed(2)} unit="%" />
                                            <ResultRow label="NDF" value={result.stats.ndf.toFixed(2)} unit="%" />
                                        </div>
                                    </div>

                                </div>
                            )}
                        </div>
                    </div>
                </div>
            </div>
        </div>
    );
}

const ResultRow = ({ label, value, unit, isWarn }) => (
    <div className="flex justify-between items-center">
        <span className="text-gray-600">{label}</span>
        <span className={`font-mono font-bold ${isWarn ? 'text-red-600' : 'text-gray-800'}`}>
            {value} <span className="text-xs font-normal text-gray-400">{unit}</span>
            {isWarn && <span className="ml-1 text-xs text-red-500">⚠</span>}
        </span>
    </div>
);
